Production Logo Implementation Syntax

The application uses the .logo--light and .logo--dark CSS classes to automatically switch logo variants when toggling between light and dark themes:

<!-- Automatic Theme-Aware Brand Logo -->
<img class="logo--light" src="{{ url_for('static', path='assets/RWML-Logo.svg') }}" alt="Reliance Weaving Mills" style="height: 38px; width: auto;">
<img class="logo--dark" src="{{ url_for('static', path='assets/RWML-Logo-Dark.svg') }}" alt="Reliance Weaving Mills" style="height: 38px; width: auto;">

  theme: {
    extend: {
      colors: {
        brand: {
          green: '#008B42',
        },
        rwml: {
          bg: 'var(--bg-page)',
          surface: 'var(--surface)',
          surfaceStrong: 'var(--surface-strong)',
          accent: 'var(--accent)',
          accentSoft: 'var(--accent-soft)',
          warm: 'var(--warm)',
